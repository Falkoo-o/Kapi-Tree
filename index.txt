        function normalize(str) {
            return str.toLowerCase().replace(/ä/g, 'ae').replace(/ö/g, 'oe').replace(/ü/g, 'ue').replace(/ß/g, 'ss').replace(/[^a-z0-9]/g, '');

                    <h3>${item.name}</h3>
                    <div class="details">
                        <div><span class="label">Gebäude:</span> ${item.geb || '-'}</div>
                        <div><span class="label">Zutaten:</span> ${item.zutaten.join(', ') || 'Keine'}</div>
                        <div><span class="label">Verkauf:</span> ${item.vk || '-'}</div>
                        <div><span class="label">Forschung:</span> ${item.ek || '-'}</div>
                        <div><span class="label">Spezial:</span> ${item.sp.join(', ') || '-'}</div>
                    </div>
                `;

                // Hover Event-Listener für das Highlighting
                card.addEventListener('mouseenter', () => highlight(item));
                card.addEventListener('mouseleave', clearHighlight);

                grid.appendChild(card);
            });
        }

        // Highlight-Logik (Rot/Grün/Blau)
        function highlight(targetItem) {
            const targetNorm = normalize(targetItem.name);
            const ingredientNorms = targetItem.zutaten.map(normalize);
            
            // Finde alle Produkte, in denen das aktuelle Produkt als Zutat benötigt wird
            const usedInNorms = rawData
                .filter(item => item.zutaten.map(normalize).includes(targetNorm))
                .map(item => normalize(item.name));

            document.querySelectorAll('.card').forEach(card => {
                const cName = card.dataset.name;

                if (cName === targetNorm) {
                    card.classList.add('active');
                } else if (ingredientNorms.includes(cName)) {
                    card.classList.add('is-ingredient'); // Rot: Input/Zutaten
                } else if (usedInNorms.includes(cName)) {
                    card.classList.add('is-product');    // Blau: Output/Weiterverarbeitung
                }
            });
        }

        function clearHighlight() {
            document.querySelectorAll('.card').forEach(card => {
                card.classList.remove('active', 'is-ingredient', 'is-building', 'is-product');
            });
        }

        // Event-Listener für Filter & Suche
        searchInput.addEventListener('input', render);
        filterGeb.addEventListener('change', render);

        // Initiales Rendern
        render();
    </script>
